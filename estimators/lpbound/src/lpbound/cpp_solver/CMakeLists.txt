cmake_minimum_required(VERSION 3.16)
project(lpbound_manifest_solver LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_CXX_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_EXTRAS_LIB OFF CACHE BOOL "" FORCE)
set(ZLIB OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/highs)
add_executable(lpbound_parallel manifest_solver.cpp)
target_include_directories(lpbound_parallel PRIVATE vendor/json)
target_link_libraries(lpbound_parallel PRIVATE highs)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(lpbound_parallel PRIVATE -O3)
endif()
